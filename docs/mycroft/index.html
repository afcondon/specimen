Functionfeed::
ParseState→String→Either String { state :: ParseState, exprs :: Array SExpr }
feed st0 chunk = foldM step { state: st0, exprs: [] } (toCharArray chunk)
  where
  step acc c = case acc.state.mode of
    Normal → normal acc c
    InString
      | c == '\\' → Right (setMode InStringEscape (consAtom acc c))
      | c == '"' → Right (setMode Normal (consAtom acc c))
      | otherwise → Right (consAtom acc c)
    InStringEscape → Right (setMode InString (consAtom acc c))
    InQuoted
      | c == '|' → Right (setMode Normal (consAtom acc c))
      | otherwise → Right (consAtom acc c)
    InComment
      | c == '\n' → Right (setMode Normal acc)
      | otherwise → Right acc

  normal acc c
    | c == '(' = Right (open (flushAtom acc))
    | c == ')' = close (flushAtom acc)
    | isSpace c = Right (flushAtom acc)
    | c == '"' = Right (setMode InString (consAtom acc c))
    | c == '|' = Right (setMode InQuoted (consAtom acc c))
    | c == ';' = Right (setMode InComment (flushAtom acc))
    | otherwise = Right (consAtom acc c)

  isSpace c = c == ' ' || c == '\n' || c == '\t' || c == '\r'

  setMode m acc = acc { state = acc.state { mode = m } }

  consAtom acc c = acc { state = acc.state { atom = c : acc.state.atom } }

  flushAtom acc = case acc.state.atom of
    Nil → acc
    cs → deliver (Atom (revString cs)) (acc { state = acc.state { atom = Nil } })

  deliver e acc = case acc.state.stack of
    Nil → acc { exprs = snoc acc.exprs e }
    top : rest → acc { state = acc.state { stack = (e : top) : rest } }

  open acc = acc { state = acc.state { stack = Nil : acc.state.stack } }

  close acc = case acc.state.stack of
    Nil → Left "unbalanced ')'"
    top : rest →
      Right
        ( deliver (List (Array.reverse (Array.fromFoldable top)))
            (acc { state = acc.state { stack = rest } })
        )
Functionfinish::ParseState → Either String (Array SExpr)
finish st = case st.mode of
  InString → Left "unterminated string literal"
  InStringEscape → Left "unterminated string literal"
  InQuoted → Left "unterminated quoted symbol"
  _ → case st.stack of
    _ : _ → Left "unterminated list"
    Nil → case st.atom of
      Nil → Right []
      cs → Right [ Atom (revString cs) ]

End of input: flush a trailing toplevel atom, reject anything left half-open.

FunctionnewSolverWith::
{ start :: Callbacks → Aff Transport , log :: Maybe (String → Effect Unit) }→Aff Solver
newSolverWith cfg = do
  refs ← liftEffect do
    parse ← Ref.new SExpr.initialState
    ready ← Ref.new []
    waiter ← Ref.new Nothing
    dead ← Ref.new Nothing
    pure { parse, ready, waiter, dead }
  let
    failWith msg = do
      d ← Ref.read refs.dead
      when (isNothing d) (Ref.write (Just msg) refs.dead)
      w ← Ref.read refs.waiter
      Ref.write Nothing refs.waiter
      for_ w \k → k (Left (error msg))

    emit e = do
      w ← Ref.read refs.waiter
      case w of
        Just k → do
          Ref.write Nothing refs.waiter
          k (Right e)
        Nothing → Ref.modify_ (_ <> [ e ]) refs.ready

    onChunk chunk = do
      for_ cfg.log \l → l ("[recv] " <> chunk)
      st ← Ref.read refs.parse
      case SExpr.feed st chunk of
        Left err → failWith ("response parse error: " <> err)
        Right r → do
          Ref.write r.state refs.parse
          for_ r.exprs emit

    onFailure msg = failWith ("solver backend: " <> msg)

  transport ← cfg.start { onChunk, onFailure }
  let
    solver = Solver
      { transport
      , parse: refs.parse
      , ready: refs.ready
      , waiter: refs.waiter
      , dead: refs.dead
      , log: cfg.log
      }
  ackCommand solver (List [ Atom "set-option", Atom ":print-success", Atom "true" ])
  pure solver

Bring up a solver over any transport and put it in lockstep mode.

Functioncommand::Solver → SExpr → Aff SExpr
command (Solver s) e = makeAff \k → do
  d ← Ref.read s.dead
  case d of
    Just msg → k (Left (error ("solver unavailable: " <> msg)))
    Nothing → do
      let line = SExpr.print e <> "\n"
      for_ s.log \l → l ("[send] " <> line)
      s.transport.send line
      buffered ← Ref.read s.ready
      case Array.uncons buffered of
        Just { head, tail } → do
          Ref.write tail s.ready
          k (Right head)
        Nothing → Ref.write (Just k) s.waiter
  pure nonCanceler

Send one command, await its one response. Lockstep discipline: callers issue one command at a time (all the veneer functions do).

FunctionnewSession::∀ claim. Solver → SmtTheory claim → Aff (Session claim)
newSession solver theory = do
  produceModels solver
  produceUnsatCores solver
  for_ theory.declare (ackCommand solver)
  for_ theory.observations \o → assertNamed solver o.name o.term
  ok ← checkSat solver >>= case _ of
    Sat → pure true
    Unsat → pure false
    Unknown → throwError (error "newSession: solver returned unknown on the observations")
  pure (Session { solver, theory, consistent: ok })
